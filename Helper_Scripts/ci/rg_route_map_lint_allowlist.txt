# shadow check only considers routes with HTTP methods (websocket routes have
# none), so a websocket-only by_path entry always reads as dead here even
# though it is real and served.

by_path /api/v1/mcp/ws* matches no served route # websocket route (mcp_unified_endpoint.py @router.websocket("/ws")); lint sees HTTP routes only
