	}

			Nonce string `json:"nonce"`
			InjectMismatch *bool `json:"inject_mismatch"`
		}

			return fmt.Errorf("decode test workspace challenge %s: %w", challengePath, err)
		}
		if len(challenge.Nonce) != 32 || challenge.InjectMismatch == nil {
			return errors.New("invalid test workspace challenge")
		}
		if *challenge.InjectMismatch {
			testWorkspaceRoot = "/workspace-mismatch/" + challenge.Nonce
		}
