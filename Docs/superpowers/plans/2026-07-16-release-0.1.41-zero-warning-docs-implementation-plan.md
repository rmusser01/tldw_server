```python
from __future__ import annotations

import hashlib
import os
import subprocess  # nosec B404 - fixed local Bash command executes a repository script
from pathlib import Path

REPO_ROOT = Path(__file__).resolve().parents[3]
REFRESH_SCRIPT = REPO_ROOT / "Helper_Scripts/refresh_docs_published.sh"


def _manifest(root: Path) -> list[tuple[str, str]]:
    return [
        (path.relative_to(root).as_posix(), hashlib.sha256(path.read_bytes()).hexdigest())
        for path in sorted(root.rglob("*"))
        if path.is_file()
    ]


def _refresh(destination: Path) -> None:
    env = os.environ.copy()
    env["TLDW_DOCS_PUBLISHED_DIR"] = str(destination)
    subprocess.run(  # nosec B603 - command is fixed to /bin/bash plus REFRESH_SCRIPT
        ["/bin/bash", str(REFRESH_SCRIPT)], cwd=REPO_ROOT, env=env,
        check=True, text=True, capture_output=True,
    )


def test_refresh_script_has_safe_destination_seam_and_no_silent_copy_failure() -> None:
    script = REFRESH_SCRIPT.read_text(encoding="utf-8")
    assert "TLDW_DOCS_PUBLISHED_DIR" in script
    assert "2>/dev/null || true" not in script
```
