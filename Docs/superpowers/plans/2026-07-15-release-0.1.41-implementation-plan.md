Immediately before starting the CI wait, run this user-authorized capacity
focus sweep. It resolves the release PR number and exact current head SHA with
`gh pr view`, paginates every queued and in-progress Actions run, preserves a
run when either its `head_sha` matches that SHA or its `pull_requests` list
contains the release PR number, and cancels only the remaining active runs.
The script prints every candidate decision and every cancellation to an audit
log, then re-enumerates until it proves only release-PR runs remain active. It
fails closed before cancellation if the PR identity is missing, malformed,
closed, merged, based anywhere other than `main`, or changes during the sweep.

Run:

```bash
set -o pipefail
PHASE=pre-wait
python - "$PHASE" <<'PY' | tee "/tmp/tldw-0.1.41-ci-focus-${PHASE}.log"
import json
import re
import subprocess
import sys
import time

REPO = "rmusser01/tldw_server"
RELEASE_BRANCH = "codex/release-main-0.1.41"
ACTIVE_STATUSES = ("queued", "in_progress")
MAX_SWEEPS = 10


def gh(*args, check=True):
    result = subprocess.run(
        ["gh", *args],
        check=False,
        capture_output=True,
        text=True,
    )
    if check and result.returncode:
        raise SystemExit(
            f"gh {' '.join(args)} failed with exit {result.returncode}: "
            f"{result.stderr.strip()}"
        )
    return result


def resolve_release_pr():
    pr = json.loads(
        gh(
            "pr", "view", RELEASE_BRANCH, "--repo", REPO,
            "--json", "number,url,headRefOid,baseRefName,state",
        ).stdout
    )
    number = pr.get("number")
    head_sha = pr.get("headRefOid")
    if not isinstance(number, int) or number <= 0:
        raise SystemExit(f"Refusing to cancel: invalid PR number: {number!r}")
    if not isinstance(head_sha, str) or not re.fullmatch(r"[0-9a-f]{40}", head_sha):
        raise SystemExit(f"Refusing to cancel: invalid headRefOid: {head_sha!r}")
    if pr.get("baseRefName") != "main" or pr.get("state") != "OPEN":
        raise SystemExit(
            "Refusing to cancel: release PR must be OPEN with base main; "
            f"got state={pr.get('state')!r}, base={pr.get('baseRefName')!r}"
        )
    return number, head_sha, pr.get("url")


def active_runs():
    by_id = {}
    for status in ACTIVE_STATUSES:
        pages = json.loads(
            gh(
                "api", "--paginate", "--slurp",
                f"repos/{REPO}/actions/runs?status={status}&per_page=100",
            ).stdout
        )
        if not isinstance(pages, list):
            raise SystemExit("Refusing to cancel: malformed paginated run response")
        for page in pages:
            runs = page.get("workflow_runs") if isinstance(page, dict) else None
            if not isinstance(runs, list):
                raise SystemExit("Refusing to cancel: malformed workflow_runs page")
            for run in runs:
                run_id = run.get("id") if isinstance(run, dict) else None
                if not isinstance(run_id, int):
                    raise SystemExit(f"Refusing to cancel: malformed run id: {run_id!r}")
                by_id[run_id] = run
    return [by_id[run_id] for run_id in sorted(by_id)]


def pull_request_numbers(run):
    pull_requests = run.get("pull_requests")
    if pull_requests is None:
        return []
    if not isinstance(pull_requests, list):
        raise SystemExit(
            f"Refusing to cancel: malformed pull_requests for run {run.get('id')}"
        )
    numbers = []
    for item in pull_requests:
        number = item.get("number") if isinstance(item, dict) else None
        if not isinstance(number, int):
            raise SystemExit(
                f"Refusing to cancel: malformed PR association for run {run.get('id')}"
            )
        numbers.append(number)
    return numbers


phase = sys.argv[1] if len(sys.argv) == 2 else ""
if phase not in {"pre-wait", "pre-merge"}:
    raise SystemExit("Usage: python - {pre-wait|pre-merge}")

release_pr = resolve_release_pr()
pr_number, head_sha, pr_url = release_pr
print(
    "RELEASE_PR "
    + json.dumps(
        {"phase": phase, "number": pr_number, "headRefOid": head_sha, "url": pr_url},
        sort_keys=True,
    )
)

cancel_requested = set()
for sweep in range(1, MAX_SWEEPS + 1):
    current_pr = resolve_release_pr()
    if current_pr[:2] != release_pr[:2]:
        raise SystemExit(
            "Refusing to cancel: release PR number or headRefOid changed during sweep"
        )

    runs = active_runs()
    classified = []
    for run in runs:
        pr_numbers = pull_request_numbers(run)
        keep = run.get("head_sha") == head_sha or pr_number in pr_numbers
        audit = {
            "decision": "KEEP_RELEASE_PR" if keep else "CANCEL_OTHER",
            "head_sha": run.get("head_sha"),
            "id": run["id"],
            "name": run.get("name"),
            "pull_requests": pr_numbers,
            "status": run.get("status"),
            "url": run.get("html_url"),
        }
        print(f"CANDIDATE sweep={sweep} " + json.dumps(audit, sort_keys=True))
        classified.append((run, keep))

    unrelated = [run for run, keep in classified if not keep]
    if not unrelated:
        print(
            "PROOF "
            + json.dumps(
                {
                    "active_release_runs": [run["id"] for run in runs],
                    "active_unrelated_runs": [],
                    "phase": phase,
                    "release_head_sha": head_sha,
                    "release_pr_number": pr_number,
                },
                sort_keys=True,
            )
        )
        break

    if resolve_release_pr()[:2] != release_pr[:2]:
        raise SystemExit(
            "Refusing to cancel: release PR changed during active-run enumeration"
        )
    for run in unrelated:
        run_id = run["id"]
        if run_id in cancel_requested:
            print(f"CANCEL_PENDING id={run_id} url={run.get('html_url')}")
            continue
        result = gh(
            "api",
            "--method",
            "POST",
            f"repos/{REPO}/actions/runs/{run_id}/cancel",
            check=False,
        )
        if result.returncode:
            raise SystemExit(
                f"Cancellation failed for run {run_id}: {result.stderr.strip()}"
            )
        cancel_requested.add(run_id)
        print(
            "CANCELLED "
            + json.dumps(
                {"id": run_id, "name": run.get("name"), "url": run.get("html_url")},
                sort_keys=True,
            )
        )
    time.sleep(2)
else:
    raise SystemExit(
        f"Post-sweep proof failed after {MAX_SWEEPS} enumerations; "
        "do not start or continue the CI wait"
    )
PY
```

number and `headRefOid`, audit-log path, candidate decisions, cancelled run
IDs/names/URLs, and final `PROOF` line. A non-zero exit blocks the wait; inspect
