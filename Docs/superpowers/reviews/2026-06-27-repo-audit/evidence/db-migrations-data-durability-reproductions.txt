db_path = root / 'app.db'
with sqlite3.connect(db_path) as conn:
    conn.execute('CREATE TABLE schema_version (version INTEGER)')
    conn.execute('INSERT INTO schema_version VALUES (0)')
    conn.commit()
migrator = DatabaseMigrator(str(db_path), str(root / 'migrations'))
migrator.initialize_migration_table()
migration = Migration(
    version=1,
    name='partial_failure_demo',
    up_sql='CREATE TABLE kept_after_failure (id INTEGER); INSERT INTO missing_table VALUES (1);',
)
try:
    migrator.execute_migration(migration)
except Exception as exc:
    print(type(exc).__name__, str(exc))
with sqlite3.connect(db_path) as conn:
    tables = [r[0] for r in conn.execute("SELECT name FROM sqlite_master WHERE type='table' ORDER BY name")]
    rows = conn.execute('SELECT version, success, error_message FROM schema_migrations ORDER BY version').fetchall()
print('tables=', tables)
print('migration_rows=', rows)
