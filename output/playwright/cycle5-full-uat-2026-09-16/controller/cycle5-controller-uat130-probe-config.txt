import base from '/Users/macbook-dev/Documents/GitHub/tldw_server2/apps/packages/ui/vitest.config'
const root = '/Users/macbook-dev/Documents/GitHub/tldw_server2/apps/packages/ui'
const testPath = root + '/src/components/Media/__tests__/AnalysisModal.model-owner.test.tsx'
const probe = `
  it.each(["qualified", "bare"])("UAT130 preserves configured identity after delayed %s catalog hydration", async (shape) => {
    const model = "../../../Working/Language_Models/Gemma/exact-model.gguf";
    const selected = "tldw:custom-openai-api:" + model;
    await new Storage().set("selectedModel", selected);
    useStoreMessageOption.setState({ selectedModel: selected });
    let resolveCatalog!: (models: any[]) => void;
    const pending = new Promise<any[]>((resolve) => { resolveCatalog = resolve });
    mocks.getChatModels.mockReturnValue(pending);
    const catalog = [
      { id: shape === "qualified" ? "ollama:gemma3:1b" : "gemma3:1b", name: "Ollama first", provider: "ollama", type: "chat" },
      { id: shape === "qualified" ? "custom-openai-api:" + model : model, name: "Configured Gemma", provider: "custom-openai-api", type: "chat" },
    ];
    function ExistingOwner() { useSelectedModel(); return null; }
    const view = render(<><ExistingOwner /><AnalysisModal open onClose={vi.fn()} mediaId={42} mediaContent="Original source" /></>);
    try {
      const select = screen.getByRole("combobox", { name: "Model" });
      expect(select.getAttribute("data-selected-value")).toBe(selected);
      await act(async () => { resolveCatalog(catalog); await pending; });
      await screen.findByRole("option", { name: "Configured Gemma" });
      const hydrated = select.getAttribute("data-selected-value");
      expect.soft([selected, "tldw:" + model]).toContain(hydrated);
      expect(useStoreMessageOption.getState().selectedModel).toBe(selected);
      expect(await new Storage().get("selectedModel")).toBe(selected);
      fireEvent.click(screen.getByRole("button", { name: "Generate Analysis" }));
      await waitFor(() => expect(mocks.bgStream).toHaveBeenCalledTimes(1));
      const body = mocks.bgStream.mock.calls[0][0].body;
      console.info("UAT130 safe identity probe", JSON.stringify({shape, hydrated, stored: useStoreMessageOption.getState().selectedModel, dispatchedModel: body.model, dispatchedProvider: body.api_provider}));
      expect.soft(body.api_provider).toBe("custom-openai-api");
      expect.soft([model, "custom-openai-api:" + model]).toContain(body.model);
      await waitFor(() => expect(mocks.messageSuccess).toHaveBeenCalled());
    } finally { view.unmount(); }
  });
`;
export default {
  ...base,
  plugins: [{ name: 'uat130-read-only-catalog-identity', enforce: 'pre' as const, transform(code: string, id: string) {
    if (id !== testPath) return null
    if (!code.includes('getChatModels: mocks.getChatModels,')) throw new Error('Catalog harness changed')
    const expanded = code.replace('getChatModels: mocks.getChatModels,', 'getChatModels: mocks.getChatModels, getModels: mocks.getChatModels,')
    const end = expanded.lastIndexOf('});')
    return { code: expanded.slice(0, end) + probe + expanded.slice(end), map: null }
  }}],
  test: { ...base.test, setupFiles: [root + '/vitest.setup.ts'], include: [testPath], testNamePattern: 'UAT130 preserves configured identity', maxWorkers: 1, fileParallelism: false },
}
