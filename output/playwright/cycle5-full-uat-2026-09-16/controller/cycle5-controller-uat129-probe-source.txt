class EmptyCollections:
    def list_prompt_collections(self, **_kwargs):
        return []
    def count_prompt_collections(self):
        return 0


async def main():
    canonical = prompts.get_auth_settings()
    print(json.dumps({"canonical_mode": canonical.AUTH_MODE, "APP_MODE_present": "APP_MODE" in os.environ,
                      "legacy_single_user_mode": prompts.settings.get("SINGLE_USER_MODE"),
                      "prompt_helper_single_user_mode": prompts._is_single_user_auth_mode()}))
    actual_get_settings = prompts.get_auth_settings
    original_legacy = prompts.settings.get("SINGLE_USER_MODE")
    calls = []
    synthetic_user = User(id=2, username="synthetic-alice", role="user", roles=["user"],
                          permissions=["prompts.read"], is_active=True, is_verified=True)
    async def controlled_unified_auth(request, **kwargs):
        calls.append(kwargs)
        if kwargs.get("token") == "synthetic-valid-session":
            return synthetic_user
        if kwargs.get("api_key") == "synthetic-valid-api-key":
            return synthetic_user
        raise HTTPException(status_code=401, detail="Invalid synthetic credential")
    prompts.get_request_user = controlled_unified_auth
    app = FastAPI()
    app.include_router(prompts.router, prefix="/api/v1/prompts")
    app.dependency_overrides[prompts.get_prompts_db_for_user] = lambda: EmptyCollections()
    rows = []
    async with AsyncClient(transport=ASGITransport(app=app), base_url="http://probe.invalid") as client:
        async def check(label, headers, expected):
            before = len(calls)
            response = await client.get("/api/v1/prompts/collections", headers=headers)
            rows.append({"case": label, "expected": expected, "actual": response.status_code,
                         "unified_auth_calls": len(calls) - before, "pass": response.status_code == expected})

        await check("canonical multi + legacy true + valid JWT", {"Authorization": "Bearer synthetic-valid-session"}, 200)
        prompts.settings["SINGLE_USER_MODE"] = False
        await check("canonical multi + legacy false + valid JWT control", {"Authorization": "Bearer synthetic-valid-session"}, 200)
        await check("multi invalid JWT remains denied", {"Authorization": "Bearer synthetic-invalid-session"}, 401)
        await check("multi missing credentials remains denied", {}, 401)
        await check("multi valid API key delegates", {"X-API-KEY": "synthetic-valid-api-key"}, 200)
        await check("multi legacy Token Bearer delegates", {"Token": "Bearer synthetic-valid-session"}, 200)
        try:
            await prompts.verify_prompts_auth(request=SimpleNamespace(), Token=None, x_api_key=None,
                                             Authorization="Bearer synthetic-valid-session")
            admin_status = 200
        except HTTPException as exc:
            admin_status = exc.status_code
        rows.append({"case": "authenticated non-admin remains rejected by admin dependency", "expected": 403,
                     "actual": admin_status, "pass": admin_status == 403})
        os.environ["PROMPTS_REQUIRE_ADMIN"] = "true"
        await check("multi optional admin gate remains enforced", {"Authorization": "Bearer synthetic-valid-session"}, 403)
        os.environ.pop("PROMPTS_REQUIRE_ADMIN")
        prompts.get_auth_settings = lambda: SimpleNamespace(AUTH_MODE="single_user", SINGLE_USER_FIXED_ID=1)
        await check("canonical single overrides legacy false; exact key works", {"X-API-KEY": "uat129-synthetic-single-key-not-real"}, 200)
        await check("canonical single wrong key denied", {"X-API-KEY": "synthetic-wrong-key"}, 401)
        prompts.get_auth_settings = actual_get_settings
        prompts.settings["SINGLE_USER_MODE"] = original_legacy
    print(json.dumps({"results": rows, "matched": sum(row["pass"] for row in rows),
                      "failed": sum(not row["pass"] for row in rows)}, indent=2))
    return 0 if all(row["pass"] for row in rows) else 1


if __name__ == "__main__":
    raise SystemExit(asyncio.run(main()))
