  it('independent: queued restore frame cannot replace newer user scroll', async () => {
    const frames: FrameRequestCallback[] = []
    globalThis.requestAnimationFrame = window.requestAnimationFrame = cb => { frames.push(cb); return frames.length }
    const container = createScrollContainer(2000, 1000, 0)
    mocks.getReadingProgress.mockResolvedValue({ media_id: 1, percent_complete: 70 })
    const scrollContainerRef = { current: container }
    renderHook(() => useMediaReadingProgress({ mediaId: 1, mediaKind: 'media', contentLength: 260, scrollContainerRef }))
    await act(async () => { await Promise.resolve() })
    expect(frames.length).toBe(1)
    act(() => {
      container.scrollTop = 400
      container.dispatchEvent(new Event('scroll'))
      frames.splice(0).forEach(cb => cb(0))
    })
    expect(container.scrollTop).toBe(400)
  })

  it('independent: replaced media invalidates queued frames without a cleanup write', async () => {
    const frames: FrameRequestCallback[] = []
    globalThis.requestAnimationFrame = window.requestAnimationFrame = cb => { frames.push(cb); return frames.length }
    const container = createScrollContainer(2000, 1000, 0)
    const scrollContainerRef = { current: container }
    mocks.getReadingProgress.mockImplementation(async id => ({ media_id: Number(id), percent_complete: id === '1' ? 70 : 20 }))
    const { rerender } = renderHook(({ mediaId }) => useMediaReadingProgress({ mediaId, mediaKind: 'media', contentLength: 260, scrollContainerRef }), { initialProps: { mediaId: 1 } })
    await act(async () => { await Promise.resolve() })
    expect(frames.length).toBe(1)
    rerender({ mediaId: 2 })
    await act(async () => { await Promise.resolve() })
    expect(frames.length).toBe(2)
    act(() => { frames[1](0); frames[0](0) })
    expect(container.scrollTop).toBe(200)
    expect(mocks.updateReadingProgress).not.toHaveBeenCalled()
  })
