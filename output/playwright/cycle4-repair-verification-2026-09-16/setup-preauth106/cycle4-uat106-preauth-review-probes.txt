  it("independent: keeps a newer durable choice when retrying a rejected pre-auth handoff write", async () => {
    mocks.config.apiKey = "";
    const original = Storage.prototype.set;
    let rejectVerifiedWrite = true;
    vi.spyOn(Storage.prototype, "set").mockImplementation(async function (this: Storage, key, value) {
      if (key === "selectedModel" && value === "tldw:custom-openai-api:gemma-tested" && rejectVerifiedWrite) {
        rejectVerifiedWrite = false;
        throw new Error("Device storage unavailable");
      }
      return original.call(this, key, value);
    });
    render(<Parent />);
    send();
