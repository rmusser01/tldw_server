it.each(["non-string-id", "unknown-node-field"])("independent: rejects malformed %s before identity mapping", async (kind) => {
  const payload = graphPayload();
  if (kind === "non-string-id") (payload.nodes[0] as unknown as {id: unknown}).id = 42;
  else Object.assign(payload.nodes[0], {unexpected: "value"});
  mocks.bgRequest.mockResolvedValueOnce(payload);
  await expect(fetchNotesGraph({centerNoteId: "alpha"})).rejects.toMatchObject({code: "notes_graph_invalid_response"});
});
it("independent: maps ordinary string IDs without prototype-key behavior", async () => {
  const payload=graphPayload();
  payload.nodes[0].id="__proto__";
  payload.nodes.push(node("constructor", "note"));
  payload.edges[0].source="__proto__";
  payload.edges.push({...payload.edges[0],id:"manual-constructor",source:"constructor",target:"__proto__",type:"manual"});
  mocks.bgRequest.mockResolvedValueOnce(payload);
  const graph=await fetchNotesGraph({centerNoteId:"__proto__"});
  expect(graph.nodes.map(n=>n.id)).toEqual(["note:__proto__","tag:amber","note:constructor"]);
  expect(graph.edges[1]).toMatchObject({source:"note:constructor",target:"note:__proto__"});
  expect(buildNotesGraphRelationshipGroups({graph,selectedNodeId:"note:__proto__",provisionalOverlays:[],suggestions:[]}).flatMap(group=>group.rows)).toHaveLength(2);
});
