
vi.mock("@/hooks/useHomeMilestoneScope", () => ({ useHomeMilestoneScope: () => "synthetic-server:single-user" }));

it.each([
  [null, "fresh browser", "tldw:custom-model"],
  ["tldw:deliberate-model", "existing deliberate selection", "tldw:deliberate-model"]
])("UAT106 wizard completion to actual model preference: %s (%s)", async (initialModel, _label, expectedModel) => {
  const { UnifiedSetupWizard } = await import("../UnifiedSetupWizard");
  const { useSelectedModel } = await import("@/hooks/chat/useSelectedModel");
  const { useStoreMessageOption } = await import("@/store/option");
  const { resolveStartupSelectedModel } = await import("@/utils/model-startup-selection");
  const { Storage } = await import("@plasmohq/storage");
  localStorage.clear();
  const storage = new Storage();
  useStoreMessageOption.setState({ selectedModel: initialModel });
  if (initialModel) await storage.set("selectedModel", initialModel);
  const models = [
    { model: "tldw:gemma3:1b", provider: "ollama", is_configured: true },
    { model: "tldw:custom-model", provider: "custom-openai-api", is_configured: true }
  ];
  setupHookMocks.verifyFirstChat.mockResolvedValueOnce({ status: "ready", provider: "custom_openai", model: "custom-model", response_text: "Hello." });
  const initial = initialStateForCompletedSteps(["setup_path", "privacy_security", "providers", "ingest_defaults", "audio_defaults", "optional_advanced", "mcp_tools"]);
  initial.step_data.providers = { acknowledged: true, default_provider: "custom_openai", default_model: "custom-model", default_provider_credential_configured: true };
  let observation: { selected: string | null, startupFallback: string | null, hydrating: boolean } | null = null;
  const ModelConsumer = () => {
    const { selectedModel, selectedModelIsLoading } = useSelectedModel();
    observation = { selected: selectedModel, hydrating: selectedModelIsLoading, startupFallback: resolveStartupSelectedModel({ currentModel: selectedModel, models, preferredModelIds: [], isCurrentModelHydrating: selectedModelIsLoading, arePreferencesHydrating: false }) };
    return <div data-testid="regular-model-preference">{selectedModel ?? "none"}</div>;
  };
  const WizardThenConsumer = () => {
    const [completed, setCompleted] = React.useState(false);
    return completed ? <ModelConsumer /> : <UnifiedSetupWizard initialState={initial} onComplete={() => setCompleted(true)} />;
  };
  const view = render(<WizardThenConsumer />);
  try {
    fireEvent.click(screen.getByRole("button", { name: /send test chat/i }));
    await screen.findByTestId("regular-model-preference");
    await waitFor(() => expect(observation?.hydrating).toBe(false));
    expect(setupHookMocks.verifyFirstChat).toHaveBeenCalledWith(expect.objectContaining({ provider: "custom_openai", model: "custom-model" }));
    expect(setupHookMocks.complete).toHaveBeenCalledTimes(1);
    const actual = observation!;
    console.log("UAT106_HANDOFF", { initialModel, verifiedProvider: "custom_openai", verifiedModel: "custom-model", selectedAfterWizard: actual.selected, startupFallback: actual.startupFallback, catalogContainsValidated: true, storedSelected: await storage.get("selectedModel") });
    expect(actual.selected ?? actual.startupFallback).toBe(expectedModel);
  } finally { view.unmount(); }
});
